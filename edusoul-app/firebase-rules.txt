rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function to check if user is authenticated
    function isAuthenticated() {
      return request.auth != null;
    }
    
    // Helper function to check if user is the owner of the document
    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }
    
    // Helper function to check if user has a specific role (checks both users and students collections)
    function hasRole(role) {
      return isAuthenticated() && 
             (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == role ||
              get(/databases/$(database)/documents/students/$(request.auth.uid)).data.role == role);
    }
    
    // Helper function to check if user is either mentor or admin
    function isMentorOrAdmin() {
      return hasRole('mentor') || hasRole('admin');
    }
    
    // Helper function to check if user is a student
    function isStudent() {
      return isAuthenticated() && 
             get(/databases/$(database)/documents/students/$(request.auth.uid)).data.role == 'student';
    }
    
    // Users collection rules (for mentors)
    match /users/{userId} {
      // Allow read if user is the owner (simple auth check, no role lookup needed)
      allow read: if isAuthenticated() && request.auth.uid == userId;
      
      // Allow create only during registration (authenticated user creating their own doc)
      allow create: if isAuthenticated() && 
                       request.auth.uid == userId &&
                       request.resource.data.role == 'mentor';
      
      // Allow update only if user is the owner
      allow update: if isAuthenticated() && request.auth.uid == userId;
      
      // Allow delete only if user is the owner
      allow delete: if isAuthenticated() && request.auth.uid == userId;
    }
    
    // Students collection rules
    match /students/{userId} {
      // Allow read if user is the owner (simple auth check, no role lookup needed)
      allow read: if isAuthenticated() && request.auth.uid == userId;
      
      // Allow create only during registration (authenticated user creating their own doc)
      allow create: if isAuthenticated() && 
                       request.auth.uid == userId &&
                       request.resource.data.role == 'student';
      
      // Allow update only if user is the owner
      allow update: if isAuthenticated() && request.auth.uid == userId;
      
      // Allow delete only if user is the owner
      allow delete: if isAuthenticated() && request.auth.uid == userId;
    }
    
    // Courses collection rules
    match /courses/{courseId} {
      // Allow read for all authenticated users
      allow read: if isAuthenticated();
      
      // Allow create only for mentors
      allow create: if hasRole('mentor') &&
                       request.resource.data.mentorId == request.auth.uid;
      
      // Allow update only for the course mentor
      allow update: if hasRole('mentor') &&
                       resource.data.mentorId == request.auth.uid;
      
      // Allow delete only for the course mentor or admin
      allow delete: if (hasRole('mentor') && resource.data.mentorId == request.auth.uid) ||
                       hasRole('admin');
    }
    
    // Enrollments subcollection under courses
    match /courses/{courseId}/enrollments/{enrollmentId} {
      // Allow read for the enrolled student, course mentor, or admin
      allow read: if isAuthenticated() && 
                     (resource.data.studentId == request.auth.uid ||
                      get(/databases/$(database)/documents/courses/$(courseId)).data.mentorId == request.auth.uid ||
                      hasRole('admin'));
      
      // Allow create for students to enroll themselves
      allow create: if hasRole('student') &&
                       request.resource.data.studentId == request.auth.uid &&
                       request.resource.data.courseId == courseId;
      
      // Allow update for the student to update their progress
      allow update: if hasRole('student') &&
                       resource.data.studentId == request.auth.uid;
      
      // Allow delete for the student to unenroll or mentor/admin
      allow delete: if (hasRole('student') && resource.data.studentId == request.auth.uid) ||
                       (hasRole('mentor') && get(/databases/$(database)/documents/courses/$(courseId)).data.mentorId == request.auth.uid) ||
                       hasRole('admin');
    }
    
    // Messages/Chats collection rules
    match /chats/{chatId} {
      // Allow read only for participants in the chat
      allow read: if isAuthenticated() && 
                     (request.auth.uid in resource.data.participants ||
                      hasRole('admin'));
      
      // Allow create if user is one of the participants
      allow create: if isAuthenticated() && 
                       request.auth.uid in request.resource.data.participants;
      
      // Allow update for participants
      allow update: if isAuthenticated() && 
                       request.auth.uid in resource.data.participants;
      
      // Allow delete for participants or admin
      allow delete: if (isAuthenticated() && request.auth.uid in resource.data.participants) ||
                       hasRole('admin');
    }
    
    // Messages subcollection under chats
    match /chats/{chatId}/messages/{messageId} {
      // Allow read for chat participants
      allow read: if isAuthenticated() && 
                     request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;
      
      // Allow create for chat participants
      allow create: if isAuthenticated() && 
                       request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants &&
                       request.resource.data.senderId == request.auth.uid;
      
      // Messages cannot be updated or deleted
      allow update, delete: if false;
    }
    
    // Reviews/Ratings collection rules
    match /reviews/{reviewId} {
      // Allow read for all authenticated users
      allow read: if isAuthenticated();
      
      // Allow create for students who are enrolled in the course
      allow create: if hasRole('student') &&
                       request.resource.data.studentId == request.auth.uid;
      
      // Allow update only by the review author
      allow update: if isAuthenticated() && 
                       resource.data.studentId == request.auth.uid;
      
      // Allow delete by the review author or admin
      allow delete: if (isAuthenticated() && resource.data.studentId == request.auth.uid) ||
                       hasRole('admin');
    }
    
    // Verifications collection rules
    match /verifications/{verificationId} {
      // Allow read for the mentor who owns the verification or admin
      allow read: if isAuthenticated() && 
                     (resource.data.mentorId == request.auth.uid || hasRole('admin'));
      
      // Allow create for mentors to save their verification
      allow create: if hasRole('mentor') &&
                       request.resource.data.mentorId == request.auth.uid;
      
      // Verifications cannot be updated or deleted (immutable records)
      allow update, delete: if false;
    }

    // CV Verifications collection rules
    match /cvVerifications/{cvVerificationId} {
      // Allow read for the mentor who owns the CV verification or admin
      allow read: if isAuthenticated() && 
                     (resource.data.mentorId == request.auth.uid || hasRole('admin'));
      
      // Allow create for mentors to upload and analyze their CV
      allow create: if hasRole('mentor') &&
                       request.resource.data.mentorId == request.auth.uid;
      
      // CV verifications cannot be updated or deleted (immutable records)
      allow update, delete: if false;
    }
    
    // Default deny rule for any other collections
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
